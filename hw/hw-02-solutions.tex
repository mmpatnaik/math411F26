\item Give the terms of order $\leq 3$ in the power series (using the expressions for $\cos(T)$, $\sin(T)$, and $e^T$ given in lecture):
\begin{enumerate}
\item $\dfrac{\sin T}{\cos T}$;
\item $\dfrac{e^T}{\sin T}$.
\end{enumerate}

\begin{solution}
Recall
\[
\sin T=T-\frac{T^3}{6}+\frac{T^5}{120}-\cdots,\quad
\cos T=1-\frac{T^2}{2}+\frac{T^4}{24}-\cdots,\quad
e^T=1+T+\frac{T^2}{2}+\frac{T^3}{6}+\frac{T^4}{24}+\cdots.
\]
Throughout, $O(T^k)$ denotes a power series all of whose terms have degree $\geq k$.

\medskip
\noindent (a) Since $\cos T$ has constant term $1\neq 0$, it is invertible in $\C[[T]]$. Write $\cos T=1-v$ with
$v=\frac{T^2}{2}-\frac{T^4}{24}+\cdots$. Since $v$ has no constant term,
\[
\frac{1}{\cos T}=1+v+v^2+\cdots=1+\frac{T^2}{2}+O(T^4),
\]
because $v^2=O(T^4)$. Therefore
\[
\frac{\sin T}{\cos T}=\Bigl(T-\frac{T^3}{6}+O(T^5)\Bigr)\Bigl(1+\frac{T^2}{2}+O(T^4)\Bigr)
=T+\Bigl(\frac12-\frac16\Bigr)T^3+O(T^5)
=\boxed{T+\frac{T^3}{3}}+O(T^5).
\]
(There is no $T^0$ or $T^2$ term, as expected since $\tan$ is odd.)

\medskip
\noindent (b) Now $\sin T$ has zero constant term, so it is \emph{not} invertible in $\C[[T]]$; the quotient
lives in the Laurent series $\C((T))$. As in lecture, factor
\[
\sin T=T\,u(T),\qquad u(T)=1-\frac{T^2}{6}+\frac{T^4}{120}-\cdots,
\]
where $u$ has constant term $1$ and is therefore invertible. Writing $u=1-w$ with
$w=\frac{T^2}{6}-\frac{T^4}{120}+\cdots$, we get
\[
\frac{1}{u}=1+w+w^2+O(T^6)
=1+\frac{T^2}{6}+\Bigl(\frac{1}{36}-\frac{1}{120}\Bigr)T^4+O(T^6)
=1+\frac{T^2}{6}+\frac{7}{360}T^4+O(T^6).
\]
Since we will divide by $T$, to get all terms of order $\leq3$ we need $e^T/u$ up to order $4$:
\begin{align*}
\frac{e^T}{u}&=\Bigl(1+T+\frac{T^2}{2}+\frac{T^3}{6}+\frac{T^4}{24}\Bigr)\Bigl(1+\frac{T^2}{6}+\frac{7}{360}T^4\Bigr)+O(T^5)\\
&=1+T+\Bigl(\frac12+\frac16\Bigr)T^2+\Bigl(\frac16+\frac16\Bigr)T^3+\Bigl(\frac1{24}+\frac1{12}+\frac{7}{360}\Bigr)T^4+O(T^5)\\
&=1+T+\frac23T^2+\frac13T^3+\frac{13}{90}T^4+O(T^5).
\end{align*}
Dividing by $T$,
\[
\frac{e^T}{\sin T}=\boxed{\frac1T+1+\frac23T+\frac13T^2+\frac{13}{90}T^3}+O(T^4).
\]
\end{solution}

\item Given complex numbers $a_0,a_1,u_1,u_2$, define $a_n$ for $n\geq 2$ by
\[
a_n=u_1a_{n-1}+u_2a_{n-2}.
\]
If
\[
T^2-u_1T-u_2=(T-\alpha)(T-\alpha')
\]
with $\alpha\neq\alpha'$, show that
\[
a_n=A\alpha^n+B(\alpha')^n
\]
for suitable $A,B$. Find $A,B$ in terms of $a_0,a_1,\alpha,\alpha'$. Consider the power series
\[
F(T)=\sum_{n=0}^{\infty}a_nT^n.
\]
Show that it represents a rational function $f(T)/g(T)$, where $f(T)$ and $g(T)$ are polynomials with $g(T)\neq 0$. Give the partial fraction decomposition of this rational function.

\begin{solution}
Expanding $(T-\alpha)(T-\alpha')$ and comparing coefficients,
\begin{equation}\label{eq:u}
u_1=\alpha+\alpha',\qquad u_2=-\alpha\alpha'.
\end{equation}
In particular $\alpha^2=u_1\alpha+u_2$ and $(\alpha')^2=u_1\alpha'+u_2$, since $\alpha,\alpha'$ are roots of $T^2-u_1T-u_2$.

\medskip
\noindent\emph{Finding $A$ and $B$.} If the formula is to hold for $n=0,1$ we need
\[
A+B=a_0,\qquad A\alpha+B\alpha'=a_1 .
\]
Since $\alpha\neq\alpha'$, this linear system has the unique solution
\begin{equation}\label{eq:AB}
A=\frac{a_1-a_0\alpha'}{\alpha-\alpha'},\qquad B=\frac{a_0\alpha-a_1}{\alpha-\alpha'}.
\end{equation}

\noindent\emph{The formula holds for all $n$.} Define $A,B$ by \eqref{eq:AB}. We show $a_n=A\alpha^n+B(\alpha')^n$ by
strong induction on $n$. The cases $n=0,1$ hold by construction. Suppose $m\geq1$ and the formula holds for all $n\leq m$. Then
\begin{align*}
a_{m+1}&=u_1a_m+u_2a_{m-1}
=u_1\bigl(A\alpha^m+B(\alpha')^m\bigr)+u_2\bigl(A\alpha^{m-1}+B(\alpha')^{m-1}\bigr)\\
&=A\alpha^{m-1}(u_1\alpha+u_2)+B(\alpha')^{m-1}(u_1\alpha'+u_2)
=A\alpha^{m+1}+B(\alpha')^{m+1},
\end{align*}
which completes the induction.

\medskip
\noindent\emph{$F$ is rational.} In $\C[[T]]$, the series $1-cT$ is invertible for any $c\in\C$, with inverse
$\sum_{n\geq0}c^nT^n$. Hence the closed form for $a_n$ gives
\[
F(T)=\sum_{n\ge0}\bigl(A\alpha^n+B(\alpha')^n\bigr)T^n
=\boxed{\frac{A}{1-\alpha T}+\frac{B}{1-\alpha' T}},
\]
with $A,B$ as in \eqref{eq:AB}. This is the partial fraction decomposition. Putting it over a common denominator and using
\eqref{eq:u},
\[
(1-\alpha T)(1-\alpha'T)=1-(\alpha+\alpha')T+\alpha\alpha'T^2=1-u_1T-u_2T^2,
\]
and
\[
A(1-\alpha'T)+B(1-\alpha T)=(A+B)-(A\alpha'+B\alpha)T=a_0+(a_1-u_1a_0)T,
\]
where we used $A\alpha'+B\alpha=(A+B)(\alpha+\alpha')-(A\alpha+B\alpha')=u_1a_0-a_1$. So
\[
F(T)=\frac{f(T)}{g(T)},\qquad f(T)=a_0+(a_1-u_1a_0)T,\qquad g(T)=1-u_1T-u_2T^2,
\]
and $g\neq0$ since its constant term is $1$.

\medskip
\noindent\emph{Remark.} One can also see $F=f/g$ directly, without the closed form. For $n\geq2$, the coefficient of $T^n$ in
$(1-u_1T-u_2T^2)F(T)$ is $a_n-u_1a_{n-1}-u_2a_{n-2}=0$. The coefficients of $T^0$ and $T^1$ are $a_0$ and $a_1-u_1a_0$.
\end{solution}

\item Determine the radius of convergence of the following series:
\begin{enumerate}
\item $\displaystyle\sum n^2z^n$;
\item $\displaystyle\sum \frac{n!}{n}z^n$;
\item If $f=\sum a_nz^n$ has radius of convergence $r>0$, show that $\sum n^da_nz^n$ has the same radius of convergence for any positive integer $d$.
\end{enumerate}

\begin{solution}
We use Hadamard's formula $1/r=\limsup_n|a_n|^{1/n}$ throughout, together with the fact that
\begin{equation}\label{eq:n1n}
\lim_{n\to\infty}n^{1/n}=1,
\end{equation}
which holds because $n^{1/n}=e^{(\log n)/n}$ and $(\log n)/n\to0$. It follows that $n^{c/n}=(n^{1/n})^c\to1$ for any fixed real $c$.

\medskip
\noindent (a) By \eqref{eq:n1n}, $(n^2)^{1/n}=(n^{1/n})^2\to1$. So $\limsup_n|n^2|^{1/n}=1$ and the radius of convergence is $\boxed{1}$.

\medskip
\noindent (b) Here $a_n=n!/n=(n-1)!$. At least half of the factors $1,2,\dots,n$ are $\geq n/2$, so $n!\geq(n/2)^{n/2}$ and
\[
(n!)^{1/n}\geq\sqrt{n/2}\longrightarrow\infty .
\]
Since $(1/n)^{1/n}\to1$ by \eqref{eq:n1n}, we have\\ $|a_n|^{1/n}=(n!)^{1/n}(1/n)^{1/n}\to\infty$. Hence
$\limsup_n|a_n|^{1/n}=\infty$ and the radius of convergence is $\boxed{0}$, i.e.\ the series converges only at $z=0$.

\medskip
\noindent (c) Set $L=\limsup_n|a_n|^{1/n}=1/r\in[0,\infty)$ and $b_n=n^{d/n}$, so that
\[
|n^da_n|^{1/n}=b_n\,|a_n|^{1/n},\qquad b_n\to1\ \text{by \eqref{eq:n1n}}.
\]
Let $0<\varepsilon<1$. For all sufficiently large $n$ we have $1-\varepsilon\leq b_n\leq1+\varepsilon$, and therefore
\[
(1-\varepsilon)|a_n|^{1/n}\leq|n^da_n|^{1/n}\leq(1+\varepsilon)|a_n|^{1/n}.
\]
Changing finitely many terms of a sequence does not change its $\limsup$, so
$(1-\varepsilon)L\leq\limsup_n|n^da_n|^{1/n}\leq(1+\varepsilon)L$. Since $\varepsilon$ was arbitrary,
$\limsup_n|n^da_n|^{1/n}=L$, and $\sum n^da_nz^n$ has radius of convergence $1/L=r$. (The same argument, with the
convention $1/0=\infty$, also covers $r=\infty$.)
\end{solution}

\item Prove the following:
\begin{enumerate}
\item The power series $\sum nz^n$ has radius of convergence $1$ but does not converge at any point of the unit circle.
\item The power series $\sum \frac{z^n}{n^2}$ has radius of convergence $1$ and converges at every point of the unit circle.
\item The power series $\sum \frac{z^n}{n}$ converges at every point of the unit circle except $z=1$.
\end{enumerate}

\begin{solution}
By \eqref{eq:n1n}, $n^{1/n}\to1$, $(1/n^2)^{1/n}\to1$ and $(1/n)^{1/n}\to1$. So by Hadamard's formula all three series
have radius of convergence $1$. It remains to study $|z|=1$. Recall that if $\sum c_n$ converges, then $c_n\to0$.

\medskip
\noindent (a) If $|z|=1$, then $|nz^n|=n\not\to0$, so $\sum nz^n$ diverges.

\medskip
\noindent (b) If $|z|=1$, then $|z^n/n^2|=1/n^2$, and $\sum1/n^2$ converges. (For instance,
$\frac1{n^2}\leq\frac1{n(n-1)}=\frac1{n-1}-\frac1n$ for $n\geq2$, and the latter series telescopes.) By the comparison test,
$\sum z^n/n^2$ converges, indeed absolutely.

\medskip
\noindent (c) At $z=1$ we get the harmonic series $\sum1/n$, which diverges. Now fix $z$ with $|z|=1$ and $z\neq1$, and set
\[
S_0=0,\qquad S_n=\sum_{k=1}^nz^k=\frac{z(1-z^n)}{1-z}\quad(n\geq1).
\]
The key point is that these partial sums are \emph{bounded}:
\[
|S_n|\leq\frac{|z|\,(1+|z|^n)}{|1-z|}=\frac{2}{|1-z|}=:M\qquad\text{for all }n.
\]
Now use \emph{summation by parts}. Since $z^n=S_n-S_{n-1}$,
\begin{align*}
\sum_{n=1}^N\frac{z^n}{n}
&=\sum_{n=1}^N\frac{S_n-S_{n-1}}{n}
=\sum_{n=1}^N\frac{S_n}{n}-\sum_{n=0}^{N-1}\frac{S_n}{n+1}\\
&=\frac{S_N}{N}+\sum_{n=1}^{N-1}S_n\Bigl(\frac1n-\frac1{n+1}\Bigr)
=\frac{S_N}{N}+\sum_{n=1}^{N-1}\frac{S_n}{n(n+1)} .
\end{align*}
As $N\to\infty$, the first term satisfies $|S_N/N|\leq M/N\to0$. For the second term,
$\bigl|S_n/(n(n+1))\bigr|\leq M/n^2$, and $\sum M/n^2$ converges. So by the comparison test,
$\sum_{n\geq1}S_n/(n(n+1))$ converges. Hence the partial sums $\sum_{n=1}^Nz^n/n$ converge as $N\to\infty$; that is,
$\sum z^n/n$ converges at every $z\neq1$ on the unit circle.
\end{solution}

\item Find the terms of order $\leq 3$ in the power series expansion of
\[
f(z)=\frac{z-2}{(z+3)(z+2)}
\]
at $z=1$.

\begin{solution}
First decompose into partial fractions. Writing $\frac{z-2}{(z+3)(z+2)}=\frac{P}{z+3}+\frac{Q}{z+2}$ and evaluating
$z-2=P(z+2)+Q(z+3)$ at $z=-3$ and $z=-2$ gives $P=5$ and $Q=-4$:
\[
f(z)=\frac{5}{z+3}-\frac{4}{z+2}.
\]
Put $w=z-1$, so that $z+3=4+w$ and $z+2=3+w$. For $a\neq0$ and $|w|<|a|$, the geometric series gives
\[
\frac{1}{a+w}=\frac1a\cdot\frac{1}{1+w/a}=\sum_{n\geq0}\frac{(-1)^n}{a^{n+1}}\,w^n .
\]
Applying this with $a=4$ and $a=3$, for $|z-1|<3$ we get
\[
f(z)=\sum_{n\geq0}c_n(z-1)^n,\qquad c_n=(-1)^n\Bigl(\frac{5}{4^{n+1}}-\frac{4}{3^{n+1}}\Bigr).
\]
Evaluating the first four coefficients:
\begin{align*}
c_0&=\tfrac54-\tfrac43=-\tfrac1{12}, &
c_1&=-\bigl(\tfrac5{16}-\tfrac49\bigr)=\tfrac{19}{144},\\
c_2&=\tfrac5{64}-\tfrac4{27}=-\tfrac{121}{1728}, &
c_3&=-\bigl(\tfrac5{256}-\tfrac4{81}\bigr)=\tfrac{619}{20736}.
\end{align*}
Hence
\[
f(z)=\boxed{-\frac1{12}+\frac{19}{144}(z-1)-\frac{121}{1728}(z-1)^2+\frac{619}{20736}(z-1)^3}+\cdots
\]
for $|z-1|<3$. The radius of convergence is exactly $3$, the distance from $1$ to the nearest pole $z=-2$.
\end{solution}

\end{enumerate}
