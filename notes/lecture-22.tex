\section{The Laurent expansion}
We begin by recalling Theorem~2.2 of Lecture~21. For
$0\leq r<R\leq\infty$, write $A(r,R)=\{z\in\C:r<|z|<R\}$.
For $\rho>0$, let $C_\rho(t)=\rho e^{it}$, $0\leq t\leq2\pi$, be the
positively oriented circle of radius $\rho$ centred at $0$. As before,
holomorphy on $\overline{A(r,R)}$ means holomorphy on an open set
containing the closed annulus, including both boundary circles.

\begin{theorem}[Laurent expansion, recalled]\label{thm:laurent}
Let $0<r<R<\infty$ and let $f$ be holomorphic on $\overline{A(r,R)}$
(that is, on an open set containing $\overline{A(r,R)}$).
Let $C_R$ and $C_r$ be the positively oriented
circles $|\zeta|=R$ and $|\zeta|=r$, and set
\[
 a_n=\frac{1}{2\pi i}\int_{C_R}\frac{f(\zeta)}{\zeta^{n+1}}\,d\zeta
 \quad(n\geq0),\qquad
 a_n=\frac{1}{2\pi i}\int_{C_r}\frac{f(\zeta)}{\zeta^{n+1}}\,d\zeta
 \quad(n<0).
\]
Then for $r<|z|<R$,
\[
 f(z)=\underbrace{\sum_{n\geq0}a_nz^n}_{f_+(z)}
 +\underbrace{\sum_{n<0}a_n\Bigl(\frac1z\Bigr)^{-n}}_{f_-(z)},
\]
and the convergence is uniform on every smaller closed annulus
$s\leq|z|\leq S$ with $r<s<S<R$.
\end{theorem}

We now prove this theorem using Cauchy's formula on an annulus from
Lecture~21, then establish uniqueness and compute expansions on different
annuli and about different centres. Finally, we use Laurent series to
classify isolated singularities.

\subsection{Proof of the Laurent expansion}\label{sec:laurent-proof}
% Handwritten Lecture 21, pages 2--3.
\begin{proof}[Proof of Theorem~\ref{thm:laurent}]
Fix $r<s<S<R$ and let $s\leq|z|\leq S$. By
Proposition~2.6 of Lecture~21,
\[
 f(z)=\underbrace{\frac{1}{2\pi i}\int_{C_R}\frac{f(\zeta)}{\zeta-z}\,d\zeta}_{(\mathrm{I})}
 \;\underbrace{-\;\frac{1}{2\pi i}\int_{C_r}\frac{f(\zeta)}{\zeta-z}\,d\zeta}_{(\mathrm{II})}.
\]
We expand the kernel $1/(\zeta-z)$ by a geometric series on each circle,
in two different ways. Let $M$ be the maximum of $|f|$ on $C_R\cup C_r$.

\textbf{The outer circle.} For $\zeta\in C_R$ we have
$|z/\zeta|\leq S/R<1$, so
\[
 \frac{1}{\zeta-z}=\frac{1}{\zeta}\,\frac{1}{1-z/\zeta}
 =\frac1\zeta\left(1+\frac z\zeta+\frac{z^2}{\zeta^2}+\cdots\right)
 =\sum_{n\geq0}\frac{z^n}{\zeta^{n+1}}.
\]
The $n$th term of $f(\zeta)z^n/\zeta^{n+1}$ is bounded by
$M R^{-1}(S/R)^n$, uniformly in $\zeta\in C_R$ and $s\leq|z|\leq S$.
Exactly as in Lecture~20 (Corollary~2.2 there), we may integrate term by
term:
\[
 (\mathrm{I})=\sum_{n\geq0}\left(\frac{1}{2\pi i}\int_{C_R}
 \frac{f(\zeta)}{\zeta^{n+1}}\,d\zeta\right)z^n=\sum_{n\geq0}a_nz^n,
\]
and the remainder after $N$ terms is at most $M(S/R)^{N+1}/(1-S/R)$.

\textbf{The inner circle.} For $\zeta\in C_r$ we instead have
$|\zeta/z|\leq r/s<1$, so
\[
 \frac{1}{\zeta-z}=-\frac1z\,\frac{1}{1-\zeta/z}
 =-\frac1z\left(1+\frac\zeta z+\frac{\zeta^2}{z^2}+\cdots\right)
 =-\sum_{m\geq0}\frac{\zeta^m}{z^{m+1}}.
\]
Now the $m$th term of $f(\zeta)\zeta^m/z^{m+1}$ is bounded by
$M s^{-1}(r/s)^m$. Integrating term by term,
\[
 (\mathrm{II})=\sum_{m\geq0}\left(\frac{1}{2\pi i}\int_{C_r}
 f(\zeta)\zeta^{m}\,d\zeta\right)z^{-(m+1)}
 =\sum_{n\leq-1}a_nz^n,
\]
where we substituted $n=-(m+1)$, so that $\zeta^m=\zeta^{-n-1}$. The
remainder after $N$ terms is at most $M(r/s)^{N+1}/(1-r/s)$.

Both remainders tend to $0$ uniformly for $s\leq|z|\leq S$, so
$f=\sum_{n\in\Z}a_nz^n$ uniformly there. Every point of $A(r,R)$ lies in
such a closed annulus.
\end{proof}

\subsection{Open annuli and uniqueness}
Combining Theorem~\ref{thm:laurent} with Lemma~2.5 of Lecture~21 gives
the version for open annuli, which is the one used in practice.

\begin{corollary}\label{cor:open-annulus}
Let $0\leq r<R\leq\infty$ and let $f$ be holomorphic on $A(r,R)$. For
$n\in\Z$ and $r<\rho<R$, the number
\[
 a_n=\frac{1}{2\pi i}\int_{C_\rho}\frac{f(\zeta)}{\zeta^{n+1}}\,d\zeta
\]
does not depend on $\rho$, and $f(z)=\sum_{n\in\Z}a_nz^n$ for all
$z\in A(r,R)$.
\end{corollary}
\begin{proof}
Independence of $\rho$ is Lemma~2.5 of Lecture~21 applied to
$h(\zeta)=f(\zeta)\zeta^{-n-1}$. Given $z\in A(r,R)$, choose
$r<\rho_1<|z|<\rho_2<R$ and apply Theorem~\ref{thm:laurent} on the closed
annulus $\rho_1\leq|z|\leq\rho_2$; its coefficients are the $a_n$ above.
\end{proof}

\begin{proposition}[Uniqueness]\label{prop:unique}
If $f(z)=\sum_{n\in\Z}c_nz^n$ on $A(r,R)$, then $c_n=a_n$ for all $n$.
\end{proposition}
\begin{proof}
Fix $k\in\Z$ and $r<\rho<R$. The series converges uniformly on $C_\rho$,
so we may multiply by $z^{-k-1}$ and integrate term by term over $C_\rho$.
Since $\int_{C_\rho}z^{m}\,dz=2\pi i$ for $m=-1$ and $0$ otherwise, only
the term $n=k$ survives:
\[
 a_k=\frac{1}{2\pi i}\int_{C_\rho}\frac{f(z)}{z^{k+1}}\,dz
 =\sum_{n\in\Z}c_n\,\frac{1}{2\pi i}\int_{C_\rho}z^{n-k-1}\,dz=c_k.\qedhere
\]
\end{proof}
In particular, \emph{any} method of producing a Laurent series for $f$
on an annulus (for example, geometric series) gives the Laurent expansion.

\section{Computing Laurent expansions}
% Handwritten Lecture 21, pages 3--4.
In practice we rarely use the coefficient integrals. Instead we use
partial fractions and geometric series, and appeal to
Proposition~\ref{prop:unique}.

\begin{example}\label{ex:zz1-at-0}
Let $f(z)=\dfrac{1}{z(z-1)}$, holomorphic on $\C\setminus\{0,1\}$. By
partial fractions,
\[
 \frac{1}{z(z-1)}=\frac{1}{z-1}-\frac1z.
\]
\begin{itemize}
 \item On $A(0,1)=\{0<|z|<1\}$: here $-1/z$ is already a Laurent series
 (valid for $|z|>0$), and $\frac{1}{z-1}=-(1+z+z^2+\cdots)$ for $|z|<1$. So
 \[
  f(z)=-\frac1z-1-z-z^2-\cdots\qquad(0<|z|<1).
 \]
 \item On $A(1,\infty)=\{|z|>1\}$: now
 $\frac{1}{z-1}=\frac1z+\frac1{z^2}+\frac1{z^3}+\cdots$
 (Example~2.3 of Lecture~21), so
 \[
  f(z)=\frac1{z^2}+\frac1{z^3}+\cdots\qquad(|z|>1).
 \]
\end{itemize}

\subsection{Laurent series about other points}
We do not need to centre Laurent series at $0$. Applying
Corollary~\ref{cor:open-annulus} to $w\mapsto f(z_0+w)$: if $f$ is
holomorphic on $\{r<|z-z_0|<R\}$, then
\[
 f(z)=\sum_{n=-\infty}^{\infty}a_n(z-z_0)^n,\qquad
 a_n=\frac{1}{2\pi i}\int_{|\zeta-z_0|=\rho}
 \frac{f(\zeta)}{(\zeta-z_0)^{n+1}}\,d\zeta\quad(r<\rho<R).
\]

\begin{example}[Expansion about $z=1$]
% Handwritten page 4 writes +(1+(1-z)+(1-z)^2+...); the sign is corrected.
Take $f(z)=\dfrac{1}{z(z-1)}=\dfrac{1}{z-1}-\dfrac1z$ again, now on
$0<|z-1|<1$. The first term is already a power of $z-1$. For the second,
\[
 \frac1z=\frac{1}{1+(z-1)}=1-(z-1)+(z-1)^2-\cdots
 =\sum_{n\geq0}(1-z)^n\qquad(|z-1|<1).
\]
Therefore the Laurent expansion about $z=1$ is
\[
 f(z)=\frac{1}{z-1}-1+(z-1)-(z-1)^2+\cdots\qquad(0<|z-1|<1).
\]
\end{example}

\begin{exercise}
Find the Laurent expansion of $\dfrac{1}{z(z-1)}$ about $z=1$ on the
region $|z-1|>1$.
\end{exercise}

\section{Isolated singularities}
% Handwritten Lecture 21, page 4. The function-side characterizations in
% the table are stated here; their proofs come in the next lecture.
\begin{definition}
Let $U$ be open and $z_0\in U$. A function $f$ holomorphic on
$U\setminus\{z_0\}$ is said to have an \emph{isolated singularity} at $z_0$.
\end{definition}
We are interested in functions $f$ on an open set that are holomorphic
except for, at worst, isolated singularities. If $f$ has an isolated
singularity at $z_0$, choose $\rho>0$ with $D(z_0,\rho)\subseteq U$. Then $f$
is holomorphic on the punctured disc $\{0<|z-z_0|<\rho\}$, an annulus with
inner radius $0$, so Corollary~\ref{cor:open-annulus} gives a Laurent
expansion
\[
 f(z)=\sum_{n=-\infty}^{\infty}a_n(z-z_0)^n\qquad(0<|z-z_0|<\rho).
\]
By Proposition~\ref{prop:unique}, the coefficients depend only on $f$ near
$z_0$, not on $\rho$.

\begin{definition}
In this situation the singularity at $z_0$ is
\begin{enumerate}
 \item \emph{removable} if $a_n=0$ for all $n<0$;
 \item a \emph{pole} if $a_n\ne0$ for some, but only finitely many, $n<0$.
 If $a_{-m}\ne0$ and $a_n=0$ for $n<-m$, the pole has \emph{order} $m$;
 \item an \emph{essential singularity} if $a_n\ne0$ for infinitely many
 $n<0$.
\end{enumerate}
\end{definition}

The three types are also visible from the behaviour of $f$ near $z_0$:

\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{|l|p{5.2cm}|p{5.4cm}|}
 \hline
 \textbf{Type} & \textbf{Behaviour of $f$ near $z_0$} &
 \textbf{Laurent expansion at $z_0$}\\
 \hline
 Removable & $f$ is bounded near $z_0$ &
 no negative terms:\newline $f=\sum_{n\geq0}a_n(z-z_0)^n$\\
 \hline
 Pole & $|f(z)|\to\infty$ as $z\to z_0$ &
 finitely many negative terms:\newline $f=\sum_{n\geq-m}a_n(z-z_0)^n$\\
 \hline
 Essential & the values of $f$ near $z_0$ are dense in $\C$ &
 infinitely many negative terms\\
 \hline
\end{tabular}
\end{center}
That the two columns match is a theorem, proved next lecture (it includes
Riemann's removable singularity theorem and the Casorati--Weierstrass
theorem).

\begin{example}
Some examples of each type:
\begin{itemize}
 \item $\dfrac{\sin z}{z}=1-\dfrac{z^2}{3!}+\dfrac{z^4}{5!}-\cdots$ has a
 removable singularity at $0$; setting its value at $0$ to be $1$ gives an
 entire function.
 \item $\dfrac{1}{z(z-1)}$ has poles of order $1$ at $0$ and at $1$, by
 the expansions computed above.
 \item $\dfrac{1}{z^2}$ has a pole of order $2$ at $0$.
 \item $e^{1/z}=\displaystyle\sum_{n\geq0}\frac{1}{n!}z^{-n}$ has an
 essential singularity at $0$.
\end{itemize}
\end{example}

\begin{exercise}
Prove the easy half of the table for the first two rows. (a) If the
Laurent expansion at $z_0$ has no negative terms, show that $f$ is bounded
near $z_0$ and extends holomorphically to $z_0$. (b) If $f$ has a pole of
order $m$ at $z_0$, write $f(z)=(z-z_0)^{-m}g(z)$ with $g$ holomorphic near
$z_0$ and $g(z_0)\ne0$, and deduce that $|f(z)|\to\infty$ as $z\to z_0$.
\end{exercise}
