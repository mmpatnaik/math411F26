We then discuss logarithm identities and compare logarithms with their

\section{Complex logarithms}
% Handwritten Lecture 18, pages 3--6.
Recall that
\[
 e^w=\sum_{n=0}^\infty\frac{w^n}{n!},\qquad
 e^{x+iy}=e^x(\cos y+i\sin y).
\]
The exponential is never zero and has period $2\pi i$. We would like to
invert it. If $z=re^{i\theta}$ with $r>0$, then all solutions of $e^w=z$ are
\[
 w=\ln r+i(\theta+2\pi k),\qquad k\in\Z.
\]
Here $\ln$ denotes the ordinary real logarithm. The difficulty is to
choose the argument continuously as $z$ varies.

\subsection{Why a branch is needed}
If we always choose $\theta\in[0,2\pi)$, the function
$\ln|z|+i\theta$ jumps across the positive real axis. At a positive
real number $r>0$, the limits from above and below are, respectively,
$\ln r$ and $\ln r+2\pi i$.

In fact, no continuous function $L$ on $\C\setminus\{0\}$ satisfies
$e^{L(z)}=z$ for all $z\ne0$. If such an $L$ existed, then
\[
 L(e^{it})-it\in2\pi i\Z\qquad(0\leq t\leq2\pi).
\]
This continuous function of $t$ takes values in a discrete set, so it
is constant. But its values at $0$ and $2\pi$ differ by $-2\pi i$,
since $e^0=e^{2\pi i}=1$. This is a contradiction.

\begin{definition}
A \emph{holomorphic branch of the logarithm} on an open set
$U\subseteq\C\setminus\{0\}$ is a holomorphic function $L\colon U\to\C$
satisfying $e^{L(z)}=z$ for every $z\in U$.
\end{definition}
Differentiating this identity gives $e^{L(z)}L'(z)=1$, hence
$L'(z)=1/z$. This suggests constructing a logarithm by integrating $1/z$.

\subsection{Constructing a logarithm on a simply connected domain}
\begin{theorem}\label{thm:log-z}
Let $U\subseteq\C\setminus\{0\}$ be a simply connected domain. Choose $z_0\in U$
and $w_0\in\C$ with $e^{w_0}=z_0$. Then
\[
 \boxed{\displaystyle L(z)=w_0+\int_{z_0}^z\frac{d\zeta}{\zeta}}
\]
defines a holomorphic branch of the logarithm on $U$, with
$L(z_0)=w_0$ and $L'(z)=1/z$.
\end{theorem}
\begin{proof}
The function $1/z$ is holomorphic on $U$, so the primitive theorem from Lecture~18
shows that the integral is path independent and $L'=1/z$.
To verify the exponential identity, differentiate:
\[
 \frac{d}{dz}\bigl(ze^{-L(z)}\bigr)
 =e^{-L(z)}-ze^{-L(z)}L'(z)=0.
\]
Since $U$ is connected, $ze^{-L(z)}$ is constant. At $z_0$, its value
is $z_0e^{-w_0}=1$, so $e^{L(z)}=z$ on $U$.
\end{proof}

\subsection{The principal branch}\label{sec:principal-log}
Take the slit plane
\[
 U_0=\C\setminus(-\infty,0],\qquad z_0=1,\qquad w_0=0.
\]
The entire nonpositive real axis, including $0$, is removed. On this
domain choose the unique argument $\Arg z\in(-\pi,\pi)$, which varies
continuously. The coordinates $z\mapsto(\ln|z|,\Arg z)$ identify $U_0$
with the convex strip $\R\times(-\pi,\pi)$; the continuous inverse is
$(x,\theta)\mapsto e^x e^{i\theta}$. Straight-line contractions in the
strip, transported back to $U_0$, contract any loop while fixing its
base point. Thus $U_0$ is simply connected.

By Theorem~\ref{thm:log-z}, the \emph{principal logarithm} is
\[
 \Log z=\int_1^z\frac{d\zeta}{\zeta},\qquad z\in U_0.
\]
Write $z=re^{i\theta}$ with $r>0$ and $-\pi<\theta<\pi$.
Use a path consisting of the positive real segment from $1$ to $r$,
followed by the circular arc from $r$ to $z$.

On the real segment, the integral is $\int_1^r dx/x=\ln r$;
this also holds when $0<r<1$, with the segment traversed towards $r$.
Parametrize the arc by $\eta(u)=re^{iu\theta}$, $0\leq u\leq1$.
This parametrization works for either sign of $\theta$, and
\[
 \int_\eta\frac{d\zeta}{\zeta}
 =\int_0^1\frac{i\theta re^{iu\theta}}{re^{iu\theta}}\,du=i\theta.
\]
Therefore
\[
 \boxed{\Log z=\ln|z|+i\Arg z,\qquad -\pi<\Arg z<\pi.}
\]
The integral construction proves that this branch is holomorphic;
the polar formula evaluates it explicitly.

\begin{warning}
The identity $\Log(zw)=\Log z+\Log w$ need not hold even when
$z,w,zw\in U_0$. For example, $z=w=e^{3\pi i/4}$ and $zw=-i$ all lie in $U_0$,
but
\[
 \Log(zw)=-\frac{\pi i}{2},\qquad \Log z+\Log w=\frac{3\pi i}{2}.
\]
The two sides differ by $2\pi i$.
\end{warning}

\section{Logarithms of nonvanishing analytic functions}
Suppose now that $f$ is \emph{analytic} on a domain $U$. Then $f'$ is
analytic, so $f'/f$ is holomorphic if $f$ has no zeros.

\begin{theorem}\label{thm:log-f}
Let $U\subseteq\C$ be a simply connected domain, and let $f$ be analytic on $U$
with $f(z)\ne0$ for all $z\in U$. Choose $z_0\in U$ and $w_0\in\C$
satisfying $e^{w_0}=f(z_0)$. Then
\[
 \boxed{\displaystyle L_f(z)=w_0+
 \int_{z_0}^z\frac{f'(\zeta)}{f(\zeta)}\,d\zeta}
\]
is holomorphic on $U$, satisfies $L_f(z_0)=w_0$, and has the properties
\[
 L_f'(z)=\frac{f'(z)}{f(z)},\qquad e^{L_f(z)}=f(z).
\]
Thus $L_f$ is a holomorphic logarithm of $f$ on $U$.
\end{theorem}
\begin{proof}
The primitive theorem from Lecture~18 applies to $f'/f$, so the integral is
path independent and gives the asserted derivative. Compute
\begin{align*}
 \frac{d}{dz}\bigl(e^{-L_f(z)}f(z)\bigr)
 &=-L_f'(z)e^{-L_f(z)}f(z)+e^{-L_f(z)}f'(z)\\
 &=e^{-L_f(z)}\left[-\frac{f'(z)}{f(z)}f(z)+f'(z)\right]=0.
\end{align*}
Since $U$ is connected, $e^{-L_f(z)}f(z)$ is constant. Evaluating at
$z_0$ gives $e^{-w_0}f(z_0)=1$. Hence $e^{L_f(z)}=f(z)$.
\end{proof}

\subsection{Dependence on the choices}
\begin{proposition}[Uniqueness up to $2\pi i\Z$]\label{prop:log-unique}
Let $U$ be connected and open. If $L_f$ and $\widetilde L_f$ are two
holomorphic logarithms of the same nonvanishing function $f$ on $U$, then
\[
 \widetilde L_f(z)-L_f(z)=2\pi i k\qquad(z\in U)
\]
for one fixed integer $k$. In particular, prescribing the value at one
point determines the logarithm uniquely.
\end{proposition}
\begin{proof}
Differentiating $e^{L_f}=e^{\widetilde L_f}=f$ shows that both derivatives
are $f'/f$. Their difference has derivative zero, so it is a constant
$C$ on $U$. Since $e^C=1$, we have $C=2\pi i k$ for some $k\in\Z$.
\end{proof}
If we change the starting point from $z_0$ to $z_1$ and choose the new
initial value to be $L_f(z_1)$, we recover exactly the same function:
\[
 L_f(z_1)+\int_{z_1}^z\frac{f'(\zeta)}{f(\zeta)}\,d\zeta=L_f(z).
\]
Choosing another allowed initial value changes the entire logarithm by
one constant multiple of $2\pi i$.

\begin{example}[Different branches on the same slit plane]
For $f(z)=z$ on $U_0=\C\setminus(-\infty,0]$, write $z=re^{i\theta}$
with $r>0$ and $-\pi<\theta<\pi$, and take $z_0=1$. The choice $w_0=0$
gives $\Log z=\ln r+i\theta$; the choice $w_0=2\pi i$ gives
\[
 L(z)=\Log z+2\pi i=\ln r+i\theta+2\pi i.
\]
Both exponentiate to $z$, and both have derivative $1/z$.
\end{example}

\begin{exercise}
Let $f$ be a nonvanishing analytic function on a simply connected domain
$U$, and let $n\geq2$ be an integer. Use a holomorphic logarithm $L_f$
to construct a holomorphic function $g$ on $U$ such that $g^n=f$.
How does changing $L_f$ by $2\pi i k$ change the root $g$?
\end{exercise}

\section{Logarithms and products}
% Handwritten Lecture 19, pages 1--2.
Do logarithms satisfy the familiar rule
$L_{f_1f_2}=L_{f_1}+L_{f_2}$? The answer depends on the initial values.
Let $f_1,f_2$ be nonvanishing analytic functions on the same simply
connected domain $U$, and fix $z_0\in U$. Choose
\[
 e^{w_1}=f_1(z_0),\qquad e^{w_2}=f_2(z_0),\qquad
 e^{w_{12}}=f_1(z_0)f_2(z_0).
\]
The corresponding integral constructions give
\begin{align*}
 L_{f_1f_2}(z)
 &=w_{12}+\int_{z_0}^z\frac{(f_1f_2)'(\zeta)}{f_1(\zeta)f_2(\zeta)}\,d\zeta\\
 &=w_{12}+\int_{z_0}^z\left(\frac{f_1'(\zeta)}{f_1(\zeta)}
                     +\frac{f_2'(\zeta)}{f_2(\zeta)}\right)\,d\zeta\\
 &=L_{f_1}(z)+L_{f_2}(z)+(w_{12}-w_1-w_2).
\end{align*}
Thus the product rule holds exactly if we choose
$w_{12}=w_1+w_2$. This is always an allowed choice, since
$e^{w_1+w_2}=f_1(z_0)f_2(z_0)$. For independently chosen branches,
$w_{12}-w_1-w_2$ may instead be a nonzero multiple of $2\pi i$.

\begin{example}[The normalized logarithm on the slit plane]
On $U_0=\C\setminus(-\infty,0]$, the choices $f(z)=z$, $z_0=1$,
and $w_0=0$ give the principal logarithm $\Log z$ from Section~\ref{sec:principal-log}.
Its normalization is $\Log 1=0$.

Prescribing principal values separately does not guarantee the product
rule: recall the example $z=w=e^{3\pi i/4}$ in Section~\ref{sec:principal-log}.
\end{example}

\section{The logarithm as a local power series}
% Handwritten Lecture 19, pages 2--3.
We know the real logarithm has a power series near $1$. Start directly
with its complex counterpart:
\[
 L_0(1+w):=\sum_{n=1}^{\infty}\frac{(-1)^{n-1}}{n}w^n.
\]
The radius of convergence is $1$, because
$\limsup_{n\to\infty}(1/n)^{1/n}=1$. For $|w|<1$, termwise
differentiation and the geometric series give
\[
 \frac{d}{dw}L_0(1+w)=\sum_{n=1}^{\infty}(-1)^{n-1}w^{n-1}
 =\frac{1}{1+w},\qquad L_0(1)=0.
\]

\subsection{Comparison with the integral construction}
Let $L$ be a holomorphic logarithm of $z$ on a domain
$U\subseteq\C\setminus\{0\}$, and fix $z_0\in U$. Put $w_0=L(z_0)$,
so $e^{w_0}=z_0$. Choose $r>0$ such that
$D(z_0,r)\subseteq U$ and $r<|z_0|$. On this disc define
\begin{align*}
 F(z)&:=w_0+L_0\!\left(1+\frac{z-z_0}{z_0}\right)\\
 &=w_0+\sum_{n=1}^{\infty}\frac{(-1)^{n-1}}{n z_0^n}(z-z_0)^n.
\end{align*}
This is analytic, and
\[
 F(z_0)=w_0,\qquad
 F'(z)=\frac{1}{z_0}\frac{1}{1+(z-z_0)/z_0}=\frac1z.
\]
But $L'(z)=1/z$ as well. Therefore $F-L$ has derivative zero on the
connected disc $D(z_0,r)$ and vanishes at $z_0$. It follows that
\[
 \boxed{\displaystyle
 L(z)=L(z_0)+\sum_{n=1}^{\infty}\frac{(-1)^{n-1}}{n z_0^n}(z-z_0)^n
 \quad (|z-z_0|<r).}
\]
In particular, every holomorphic branch of the logarithm is analytic.
This uses the explicit power series; the general theorem that
holomorphic functions are analytic has yet to be proved.

\subsection{The picture of overlapping discs}
The picture shows local logarithm series on discs avoiding $0$ along a
circle around $0$. On each new disc, adjust the initial value by a
multiple of $2\pi i$ to match the preceding branch on the overlap.
The overlap is convex, hence connected. Since both derivatives are $1/z$,
matching at one point makes the branches agree throughout the overlap.

Why does continuing the series once around $0$ change its value?
Parametrize the circle by $z(t)=e^{it}$, $0\leq t\leq2\pi$, and choose
\[
 0=t_0<t_1<\cdots<t_N=2\pi
\]
so that the $j$-th arc lies in a disc $D_j$ avoiding $0$. Let $L_j$ be
the local logarithm series used on that arc. Differentiating each series gives
$L_j'(z)=1/z$, so the chain rule gives
\[
 \frac{d}{dt}L_j(e^{it})=\frac{1}{e^{it}}\,ie^{it}=i.
\]
Thus $L_j(e^{it})-it$ has derivative zero and is constant for
$t_{j-1}\leq t\leq t_j$. Evaluating this constant at the two endpoints
gives
\[
 L_j(e^{it_j})-it_j
 =L_j(e^{it_{j-1}})-it_{j-1}.
\]
Rearranging, the change along that arc is
\[
 L_j(e^{it_j})-L_j(e^{it_{j-1}})=i(t_j-t_{j-1}).
\]
This calculation uses the same branch $L_j$ throughout the arc.
Matching different branches enters only when we pass to the next arc.

At each transition we choose the next branch to agree with the previous
one on their overlap; in particular,
$L_{j+1}(e^{it_j})=L_j(e^{it_j})$.

To see the cancellation explicitly, put $z_j=e^{it_j}$, so that
$z_0=z_N=1$. Adding the changes along the individual arcs gives
\begin{align*}
 \sum_{j=1}^N\bigl(L_j(z_j)-L_j(z_{j-1})\bigr)
 &=\bigl[L_1(z_1)-L_1(z_0)\bigr]
   +\bigl[L_2(z_2)-L_2(z_1)\bigr]\\
 &\quad+\cdots+\bigl[L_N(z_N)-L_N(z_{N-1})\bigr].
\end{align*}
The term $+L_1(z_1)$ from the first arc cancels with $-L_2(z_1)$
from the second arc because the branches were matched at $z_1$:
$L_1(z_1)=L_2(z_1)$. Similarly, at every intermediate point $z_j$,
$L_j(z_j)-L_{j+1}(z_j)=0$ for $1\leq j\leq N-1$.
More explicitly, regrouping the sum gives
\begin{align*}
 \sum_{j=1}^N\bigl(L_j(z_j)-L_j(z_{j-1})\bigr)
 &=L_N(z_N)-L_1(z_0)
   +\sum_{j=1}^{N-1}\underbrace{\bigl(L_j(z_j)-L_{j+1}(z_j)\bigr)}_{=\,0}\\
 &=L_N(1)-L_1(1).
\end{align*}
Only the final value and the initial value remain. Although both are
evaluated at the point $1$, they belong to different branches, $L_N$
and $L_1$; we have not imposed a matching condition between these two.
Since the change along the $j$-th arc is $i(t_j-t_{j-1})$, we obtain
\[
 \boxed{L_N(1)-L_1(1)
 =\sum_{j=1}^N i(t_j-t_{j-1})=2\pi i.}
\]
Thus a starting value $L_1(1)=0$ becomes $L_N(1)=2\pi i$ after one
counterclockwise turn. Equivalently, the change on each arc is
$\int_{\text{arc}}dz/z$, so the total change is
$\int_{|z|=1}dz/z=2\pi i$. The integral records the accumulated changes
of the matched local series.

\textbf{Why do the last and first branches not agree on their overlap?}
Two local logarithm series need not agree on overlapping discs; their
difference on the overlap is a constant $2\pi i k$, $k\in\Z$. Matching
each new branch with the preceding one fixes this constant. By the time
we return to the first disc, the final branch is fixed, and the calculation
above shows that it differs from the initial branch by $2\pi i$ on their
overlap. Subtracting $2\pi i$ from the final branch would make it agree
with the first, but would break its agreement with the preceding branch.

We can match successive branches along the path, but cannot make all
these matches consistent around the loop. Thus local logarithm series
do not give a single-valued logarithm on all of $\C\setminus\{0\}$.
Simple connectedness removes this obstruction in the integral construction.
